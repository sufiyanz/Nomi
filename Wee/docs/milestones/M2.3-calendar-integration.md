## iOS Calendar Integration

### EventKit Setup
```swift
import EventKit

class EventKitManager {
    private let eventStore = EKEventStore()
    
    func requestAccess() async throws -> Bool {
        let status = EKEventStore.authorizationStatus(for: .event)
        
        switch status {
        case .authorized:
            return true
        case .notDetermined:
            return try await eventStore.requestAccess(to: .event)
        case .denied, .restricted:
            throw CalendarError.accessDenied
        @unknown default:
            throw CalendarError.unknownAuthorizationStatus
        }
    }
    
    func fetchCalendars() -> [EKCalendar] {
        return eventStore.calendars(for: .event)
    }
}
```

## Error Handling

### Calendar Errors
```swift
enum CalendarError: LocalizedError {
    case accessDenied
    case networkUnavailable
    case authenticationFailed
    case syncConflict([SyncConflict])
    case rateLimitExceeded
    case serviceUnavailable
    case unknownAuthorizationStatus
    
    var errorDescription: String? {
        switch self {
        case .accessDenied:
            return "Calendar access denied. Please enable in Settings."
        case .networkUnavailable:
            return "Network connection required for sync."
        case .authenticationFailed:
            return "Authentication failed. Please sign in again."
        case .syncConflict(let conflicts):
            return "Sync conflicts detected for \(conflicts.count) events."
        case .rateLimitExceeded:
            return "Too many requests. Please try again later."
        case .serviceUnavailable:
            return "Calendar service temporarily unavailable."
        case .unknownAuthorizationStatus:
            return "Unknown calendar authorization status."
        }
    }
}
```

### Error Recovery
```swift
class ErrorRecoveryManager {
    func handleCalendarError(_ error: CalendarError) {
        switch error {
        case .accessDenied:
            showCalendarPermissionDialog()
        case .authenticationFailed:
            initiateReauthentication()
        case .syncConflict(let conflicts):
            showConflictResolutionDialog(conflicts)
        case .networkUnavailable:
            enableOfflineMode()
        case .rateLimitExceeded:
            scheduleRetryAfterDelay()
        case .serviceUnavailable:
            fallbackToLocalCalendar()
        case .unknownAuthorizationStatus:
            requestCalendarPermission()
        }
    }
}
```
